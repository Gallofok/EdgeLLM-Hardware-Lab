cmake_minimum_required(VERSION 3.16)
project(EdgeLLMGGMLFeasibility LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(EDGELLM_HARDWARE_LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(EDGELLM_Q5_HYBRID "Build the multi-vector hybrid dispatch candidate" OFF)
if(EDGELLM_Q5_HYBRID)
  set(edgellm_default_source "${EDGELLM_HARDWARE_LAB_ROOT}/vendor/llama-row4-hybrid")
else()
  set(edgellm_default_source "${EDGELLM_HARDWARE_LAB_ROOT}/vendor/llama-row4")
endif()
set(EDGELLM_LLAMA_SOURCE "${edgellm_default_source}" CACHE PATH "Isolated, prepared pinned engine")
# The hybrid candidate needs its own engine patch; refuse a mismatched source.
file(READ "${EDGELLM_LLAMA_SOURCE}/ggml/src/ggml-cpu/ggml-cpu.c" edgellm_engine_source)
string(FIND "${edgellm_engine_source}" "edgellm_q5_try_row4_vectors" edgellm_hybrid_patch)
if(EDGELLM_Q5_HYBRID AND edgellm_hybrid_patch EQUAL -1)
  message(FATAL_ERROR "EDGELLM_Q5_HYBRID requires the engine prepared with the hybrid patch")
elseif(NOT EDGELLM_Q5_HYBRID AND NOT edgellm_hybrid_patch EQUAL -1)
  message(FATAL_ERROR "The hybrid-patched engine requires EDGELLM_Q5_HYBRID")
endif()
unset(edgellm_engine_source)
option(GGML_EDGELLM_Q5_ROW4 "Experimental Q5 row4 integration (ARM64 only)" OFF)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
option(EDGELLM_BUILD_LLAMA_BENCH "Build a driver around the pinned official benchmark" OFF)
set(LLAMA_BUILD_COMMON ${EDGELLM_BUILD_LLAMA_BENCH} CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${EDGELLM_LLAMA_SOURCE}" engine)
option(EDGELLM_Q5_TWO_BLOCK "Build the two-block compatible loop candidate" OFF)
option(EDGELLM_Q5_TWO_ROW "Build the two-row prefetching loop candidate" OFF)
option(EDGELLM_Q5_TWO_ROW_PLAIN "Build the two-row loop candidate without prefetch hints" OFF)
set(edgellm_candidates 0)
foreach(edgellm_option EDGELLM_Q5_TWO_BLOCK EDGELLM_Q5_HYBRID EDGELLM_Q5_TWO_ROW
    EDGELLM_Q5_TWO_ROW_PLAIN)
  if(${edgellm_option})
    math(EXPR edgellm_candidates "${edgellm_candidates} + 1")
  endif()
endforeach()
if(edgellm_candidates GREATER 1)
  message(FATAL_ERROR "Select one candidate; the hybrid already compiles the two-block loop")
endif()
if(edgellm_candidates EQUAL 1)
  if(NOT GGML_EDGELLM_Q5_ROW4)
    message(FATAL_ERROR "Candidate builds require GGML_EDGELLM_Q5_ROW4")
  endif()
  # The measured bridge and reviewed engine patch stay byte-identical.
  set(edgellm_bridge "${EDGELLM_HARDWARE_LAB_ROOT}/src/integration/q5_ggml_bridge.cpp")
  if(EDGELLM_Q5_HYBRID)
    set(edgellm_candidate "${EDGELLM_HARDWARE_LAB_ROOT}/src/integration/q5_ggml_bridge_hybrid.cpp")
  elseif(EDGELLM_Q5_TWO_ROW)
    set(edgellm_candidate "${EDGELLM_HARDWARE_LAB_ROOT}/src/integration/q5_ggml_bridge_two_row.cpp")
  elseif(EDGELLM_Q5_TWO_ROW_PLAIN)
    set(edgellm_candidate
      "${EDGELLM_HARDWARE_LAB_ROOT}/src/integration/q5_ggml_bridge_two_row_plain.cpp")
  else()
    set(edgellm_candidate "${EDGELLM_HARDWARE_LAB_ROOT}/src/integration/q5_ggml_bridge_two_block.cpp")
  endif()
  get_target_property(edgellm_sources ggml-cpu SOURCES)
  if(NOT edgellm_bridge IN_LIST edgellm_sources)
    message(FATAL_ERROR "The measured bridge is not a ggml-cpu source")
  endif()
  list(TRANSFORM edgellm_sources REPLACE "^${edgellm_bridge}$" "${edgellm_candidate}")
  set_target_properties(ggml-cpu PROPERTIES SOURCES "${edgellm_sources}")
  set_source_files_properties("${edgellm_candidate}" TARGET_DIRECTORY ggml-cpu
    PROPERTIES COMPILE_OPTIONS "-march=armv8-a;-fno-fast-math;-ffp-contract=off")
endif()
enable_testing()
add_executable(test-ggml-row4 "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/integration/test_ggml_row4.cpp")
target_include_directories(test-ggml-row4 PRIVATE "${EDGELLM_HARDWARE_LAB_ROOT}/src" "${EDGELLM_HARDWARE_LAB_ROOT}/benchmarks")
target_link_libraries(test-ggml-row4 PRIVATE ggml ggml-cpu ${CMAKE_DL_LIBS})
target_compile_options(test-ggml-row4 PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
if(GGML_EDGELLM_Q5_ROW4)
  target_compile_definitions(test-ggml-row4 PRIVATE EDGELLM_INTEGRATED=1)
endif()
if(EDGELLM_Q5_HYBRID)
  # The measured harnesses expect every eligible call to reach the candidate.
  foreach(edgellm_harness ggml model)
    add_executable(test-${edgellm_harness}-row4-hybrid
      "${EDGELLM_HARDWARE_LAB_ROOT}/tests/integration/test_${edgellm_harness}_row4_hybrid.cpp")
    target_include_directories(test-${edgellm_harness}-row4-hybrid PRIVATE
      "${EDGELLM_HARDWARE_LAB_ROOT}/src" "${EDGELLM_HARDWARE_LAB_ROOT}/benchmarks")
    target_compile_options(test-${edgellm_harness}-row4-hybrid PRIVATE
      -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
    target_compile_definitions(test-${edgellm_harness}-row4-hybrid PRIVATE EDGELLM_INTEGRATED=1)
  endforeach()
  target_link_libraries(test-ggml-row4-hybrid PRIVATE ggml ggml-cpu ${CMAKE_DL_LIBS})
  target_link_libraries(test-model-row4-hybrid PRIVATE llama ggml-cpu)
  add_test(NAME ggml-row4-hybrid-dispatch COMMAND test-ggml-row4-hybrid)
else()
  add_test(NAME ggml-row4-layout-threads COMMAND test-ggml-row4)
endif()
add_executable(test-model-row4 "${EDGELLM_HARDWARE_LAB_ROOT}/tests/integration/test_model_row4.cpp")
target_include_directories(test-model-row4 PRIVATE "${EDGELLM_HARDWARE_LAB_ROOT}/src")
target_link_libraries(test-model-row4 PRIVATE llama ggml-cpu)
target_compile_options(test-model-row4 PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
if(GGML_EDGELLM_Q5_ROW4)
  target_compile_definitions(test-model-row4 PRIVATE EDGELLM_INTEGRATED=1)
endif()

if(EDGELLM_BUILD_LLAMA_BENCH)
  if(NOT GGML_EDGELLM_Q5_ROW4)
    message(FATAL_ERROR "The benchmark driver requires the opt-in ARM64 integration build")
  endif()
  add_executable(edgellm-llama-bench "${EDGELLM_HARDWARE_LAB_ROOT}/benchmarks/llama_bench_driver.cpp")
  target_compile_definitions(edgellm-llama-bench PRIVATE
    EDGELLM_LLAMA_BENCH_SOURCE="${EDGELLM_LLAMA_SOURCE}/tools/llama-bench/llama-bench.cpp")
  if(EDGELLM_Q5_HYBRID)
    target_compile_definitions(edgellm-llama-bench PRIVATE EDGELLM_Q5_HYBRID=1)
  endif()
  target_include_directories(edgellm-llama-bench PRIVATE "${EDGELLM_HARDWARE_LAB_ROOT}/src")
  target_link_libraries(edgellm-llama-bench PRIVATE common llama ggml-cpu ${CMAKE_THREAD_LIBS_INIT})
endif()

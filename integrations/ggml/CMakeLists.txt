cmake_minimum_required(VERSION 3.16)
project(EdgeLLMGGMLFeasibility LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(EDGELLM_HARDWARE_LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EDGELLM_LLAMA_SOURCE "${EDGELLM_HARDWARE_LAB_ROOT}/vendor/llama-row4" CACHE PATH "Isolated, prepared pinned engine")
option(GGML_EDGELLM_Q5_ROW4 "Experimental Q5 row4 integration (ARM64 only)" OFF)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${EDGELLM_LLAMA_SOURCE}" engine)
enable_testing()
add_executable(test-ggml-row4 "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/integration/test_ggml_row4.cpp")
target_include_directories(test-ggml-row4 PRIVATE "${EDGELLM_HARDWARE_LAB_ROOT}/src" "${EDGELLM_HARDWARE_LAB_ROOT}/benchmarks")
target_link_libraries(test-ggml-row4 PRIVATE ggml ggml-cpu ${CMAKE_DL_LIBS})
target_compile_options(test-ggml-row4 PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
if(GGML_EDGELLM_Q5_ROW4)
  target_compile_definitions(test-ggml-row4 PRIVATE EDGELLM_INTEGRATED=1)
endif()
add_test(NAME ggml-row4-layout-threads COMMAND test-ggml-row4)
add_executable(test-model-row4 "${EDGELLM_HARDWARE_LAB_ROOT}/tests/integration/test_model_row4.cpp")
target_include_directories(test-model-row4 PRIVATE "${EDGELLM_HARDWARE_LAB_ROOT}/src")
target_link_libraries(test-model-row4 PRIVATE llama ggml-cpu)
target_compile_options(test-model-row4 PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
if(GGML_EDGELLM_Q5_ROW4)
  target_compile_definitions(test-model-row4 PRIVATE EDGELLM_INTEGRATED=1)
endif()
